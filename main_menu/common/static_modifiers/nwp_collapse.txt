# New World Pestilence - carrying-capacity collapse.
#
# Half A of the design: the thing that stops the rebound. Disease is
# subtraction; EU5's free-land growth brackets are a spring, and emptying a
# location is what arms them. This models why the land could not carry the old
# densities again for a century and a half:
#
#   - chinampas, terraces and irrigation stopped being maintained because the
#     maintainers were dead
#   - congregacion uprooted survivors from land they knew how to farm
#   - encomienda and repartimiento pulled labour into mines and estates
#   - Spanish livestock took the rest (Melville's "plague of sheep" on the
#     Valle del Mezquital)
#
# New modifier keys, so no REPLACE:/INJECT: prefix. Definition idiom copied
# from M&T's main_menu/common/static_modifiers/MnT_centers.txt.
#
# Applied by the monthly sweeps in in_game/common/situations/
# nwp_demographic_collapse.txt, stamped for 60 years and gated on region
# contact, with new stamps stopping at 1600.

# CAPACITY IS THE LOAD-BEARING FIELD HERE. Measured 1676: Mesoamerica holds
# 19.8M across 325 locations at 0.610 fill, so ~100k capacity per location -
# the same as Britain, Iberia and Colombia. Capacity is not the anomaly; the
# population is.
#
# But at any target in the 1-3M band, pop/location falls under 10k AND under
# 10% of capacity, which is both conditions for abundant_free_land and its
# +1.5%/yr. A 2M Mesoamerica doubles in 46 years. Every historically correct
# outcome arms the spring, precisely because the mortality worked.
#
# Cutting capacity is what fails the fill condition and keeps the collapsed
# population out of that bracket:
#
#   floor   cap/loc must be under   cut from ~100k
#   1.0M            30.8k                69%
#   1.5M            46.2k                54%
#   2.0M            61.5k                38%
#
# The two modifiers below total -65% additively or -55% multiplicatively,
# supporting a floor of roughly 1.0M or 1.5M. Which of those applies is
# UNTESTED and moves the supportable floor by half a million.
#
# DURATION: stamped for 60 years, not 100. Note the interaction with the 1600
# cutoff on new stamps - colonial disruption re-stamps on expiry while a
# location stays native-majority, so the effective clearing is:
#
#   stamped 1493 -> expires 1553 -> re-stamped -> expires 1613 -> clears
#   stamped just before 1600 -> clears 1659
#
# So suppression drains across roughly 1613-1659 rather than 1693-1699 under
# the old 100-year stamps, leaving 41-87 years of recovery runway before 1700.
#
# Applied for decades after a wave passes: the fields are still there, the
# people who knew how to work them are not.
nwp_agricultural_collapse = {
	game_data = {
		category = location
	}
	local_population_growth = -0.010	# was -0.005
	local_population_capacity_modifier = -0.45	# was -0.25
	local_monthly_development = -0.004
	local_prosperity_decay = 0.15

	# The real weapon. Vanilla's province_starving is untouched by M&T and
	# carries local_population_growth = -0.025 - roughly 17x Abundant Free
	# Land's +1.5%. We do not need to out-tune the growth spring; we need to
	# stop the location feeding itself and let vanilla's own starvation
	# mechanism do the work. Cf. scorched_earth (-0.5), russian_famine (-0.75).
	local_monthly_food_modifier = -0.35
}

# The heavier version, for locations under an Old World overlord running
# encomienda-style extraction. Can push a location net negative on its own.
nwp_colonial_labour_regime = {
	game_data = {
		category = location
	}
	# -0.012 -> -0.009 to make room for the disease.
	#
	# At -0.012 this modifier, on 1,793 locations, was producing the ENTIRE
	# -67.9% at 1565 by itself, because cocoliztli never spawned and
	# nwp_agricultural_collapse is disease-gated (frozen at 23 of 325 in
	# Mesoamerica). Fixing the spawn gate brings agricultural collapse and the
	# disease's own -0.05 location modifier online across most of the region,
	# so labour regime no longer has to carry it alone.
	#
	# Deliberately a modest cut: the brief is to land under the 1-3M target
	# rather than risk missing it high. Rough decomposition of the measured
	# -1.382%/yr suggests this lands near -1.7 to -2.0%/yr sustained, before
	# epidemic mortality spikes - roughly 1.8-2.3M at 1600.
	local_population_growth = -0.009	# was -0.008, then -0.012
	local_population_capacity_modifier = -0.40
	local_monthly_development = -0.006
	local_prosperity_decay = 0.25
	local_migration_attraction = -15.0
	local_monthly_food_modifier = -0.5
}


# Virgin-soil susceptibility.
#
# `local_disease_resistance` is a real modifier type, and vanilla already uses
# it for exactly this purpose: `spa_severe_smallpox_outbreak` carries
# -0.75, `spa_mild_smallpox_outbreak` -0.25 (see
# docs/vanilla-reference/location.txt). M&T uses the positive direction on
# town buildings (hospitals etc., +0.1 to +0.33).
#
# This is the cleanest statement of the historical fact: New World pops had no
# prior exposure to Old World crowd diseases. Applying it to American locations
# makes the EXISTING smallpox/measles/influenza far deadlier there without
# editing their definitions, which would change them globally and break the
# Old World balance M&T tuned them for.
#
# Applied and lifted every month by the Situation sweep, keyed on
# dominant_culture: a location that has become settler-majority stops being
# virgin soil. Measured at ~4,300 locations from 1536 on.
nwp_virgin_soil = {
	game_data = {
		category = location
	}
	local_disease_resistance = -0.75
	local_life_expectancy = -10
}
