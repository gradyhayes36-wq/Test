namespace = nwp

# The Cortes expedition.
#
# WHY THIS EXISTS: in a 12-hour unsteered observer run the AI never took a
# single Mesoamerican location. Nothing in the submod is eligible to fire in a
# region no Old World power has reached, so the whole collapse model sat idle
# over the one region it was built for. EU5's Conquistador mechanic would
# normally cover this; M&T has it disabled.
#
# This forces the historical foothold - Veracruz, 1519, launched from the
# Caribbean - and lets the existing machinery take it from there. It is
# deliberately the SMALLEST possible intervention: one location, once. Nothing
# here conquers Mexico. It gives Spain the beachhead and leaves the rest to
# the AI, the diseases and the region-contact sweep.
#
# It is tag-driven on Spain but player-agnostic: it fires the same whoever the
# human is playing, including observer mode.
#
# ============================ UNVERIFIED ============================
# Three tokens below could not be checked against the repo's vanilla
# reference files, which contain only modifier definitions and situations:
#
#   1. set_owner              - the ownership-transfer effect name
#   2. c:SPA                  - Spain's tag (may be CAS/Castile at this date)
#   3. cuetlaxtlan_province   - province key, and whether it scopes as
#                               province: or location:
#
# nwp.21 and nwp.22 exist to settle all three from the console before any of
# this is wired into the monthly sweep. Do not enable nwp.20 until one of them
# is confirmed working.
# ====================================================================


# --------------------------------------------------------------------
# nwp.21 - DEBUG. Grant Veracruz to Spain, treating the key as a PROVINCE.
# Console: event nwp.21
# --------------------------------------------------------------------
nwp.21 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		province:cuetlaxtlan_province ?= {
			every_location_in_province = {
				set_owner = c:SPA
			}
			error_log = "::NWPCORTES::[GetCurrentYear]:DEBUG province-scope grant attempted"
		}
	}
}


# --------------------------------------------------------------------
# nwp.22 - DEBUG. Same, treating the key as a LOCATION.
# Console: event nwp.22
# --------------------------------------------------------------------
nwp.22 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		location:cuetlaxtlan_province ?= {
			set_owner = c:SPA
			error_log = "::NWPCORTES::[GetCurrentYear]:DEBUG location-scope grant attempted"
		}
	}
}
