namespace = nwp

# The Cortes expedition.
#
# WHY THIS EXISTS: in a 12-hour unsteered observer run the AI never took a
# single Mesoamerican location. Nothing in the submod is eligible to fire in a
# region no Old World power has reached, so the whole collapse model sat idle
# over the one region it was built for. EU5's Conquistador mechanic would
# normally cover this; M&T has it disabled.
#
# This forces the historical foothold - Veracruz, 1519, launched from the
# Caribbean - and lets the existing machinery take it from there. It is
# deliberately the SMALLEST possible intervention: one location, once. Nothing
# here conquers Mexico. It gives the beachhead and leaves the rest to the AI,
# the diseases and the region-contact sweep.
#
# SPONSOR: whoever owns Sevilla, not a hardcoded Spain. Historically the same
# thing - the Indies trade was run from Seville under the Casa de
# Contratacion, and Cortes sailed on Castilian authority - but it degrades
# into alternate history gracefully. If Morocco retakes Andalusia before the
# expedition launches, Morocco gets Veracruz. It also dodges the question of
# whether Spain exists as a tag yet at this date.
#
# Player-agnostic either way: it fires the same whoever the human is playing,
# including observer mode.
#
# ============================ UNVERIFIED ============================
# Four tokens below could not be checked against the repo's vanilla reference
# files, which contain only modifier definitions and situations:
#
#   1. set_owner              - the ownership-transfer effect name
#   2. location:sevilla       - Seville's location key
#   3. owner                  - location -> country scope transition
#   4. cuetlaxtlan_province   - province key, and whether it scopes as
#                               province: or location:
#
# The three probes below settle all four from the console before any of this
# is wired into the monthly sweep. Do not enable nwp.20 until they pass.
# ====================================================================


# --------------------------------------------------------------------
# nwp.23 - DEBUG. Resolve the sponsor and report its New World holdings.
# Console: event nwp.23
#
# Tests tokens 2 and 3, and the real trigger condition, in one command.
# every_owned_location is already proven - nwp.1 uses it successfully.
# --------------------------------------------------------------------
nwp.23 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		location:sevilla ?= {
			owner ?= {
				save_scope_as = cortes_sponsor
				# MEASURED 1337: this resolved, and reported
				# new_world_locations=0 even with a Castilian colony in Cuba
				# on the map - and in the same log the Situation's can_start
				# fired, which proves SOME Old World power owns a populated
				# American location. So either Sevilla's owner is not the
				# country holding Cuba, or the colony belongs to a subject.
				#
				# total_locations separates those: a Castile-sized number
				# means the owner is right and the American filter is wrong;
				# a handful means Sevilla belongs to someone unexpected.
				set_local_variable = {
					name = nwp_all
					value = { every_owned_location = { add = 1 } }
				}
				set_local_variable = {
					name = nwp_nw
					value = {
						every_owned_location = {
							limit = { continent = continent:america }
							add = 1
						}
					}
				}
				error_log = "::NWPCORTES::[GetCurrentYear]:Sevilla owner resolved:total_locations=[SCOPE.GetLocalVariable('nwp_all').GetValue]:new_world_locations=[SCOPE.GetLocalVariable('nwp_nw').GetValue]"
				remove_local_variable = nwp_all
				remove_local_variable = nwp_nw
			}
		}
	}
}


# --------------------------------------------------------------------
# nwp.24 - DEBUG. Which key form does Veracruz actually answer to?
# Console: event nwp.24
#
# MEASURED 1337: neither nwp.21 nor nwp.22 logged anything, and their
# error_logs sit INSIDE the scope block - so cuetlaxtlan_province resolved
# as neither a province nor a location. The in-game name is Xamapan, not
# Veracruz, so the key is probably built on that instead.
#
# This grants nothing. Every candidate is ?= guarded with its own log line,
# so one console command says which form exists.
# --------------------------------------------------------------------
nwp.24 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		location:xamapan ?= {
			error_log = "::NWPKEY::HIT location:xamapan"
		}
		location:xamapan_location ?= {
			error_log = "::NWPKEY::HIT location:xamapan_location"
		}
		province:xamapan_province ?= {
			error_log = "::NWPKEY::HIT province:xamapan_province"
		}
		location:cuetlaxtlan ?= {
			error_log = "::NWPKEY::HIT location:cuetlaxtlan"
		}
		province:cuetlaxtlan ?= {
			error_log = "::NWPKEY::HIT province:cuetlaxtlan"
		}
		error_log = "::NWPKEY::probe complete"
	}
}


# --------------------------------------------------------------------
# nwp.21 - DEBUG. Grant Veracruz to Sevilla's owner, key read as a PROVINCE.
# Console: event nwp.21
# --------------------------------------------------------------------
nwp.21 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		location:sevilla ?= {
			owner ?= { save_scope_as = cortes_sponsor }
		}
		province:cuetlaxtlan_province ?= {
			every_location_in_province = {
				set_owner = scope:cortes_sponsor
			}
			error_log = "::NWPCORTES::[GetCurrentYear]:DEBUG province-scope grant attempted"
		}
	}
}


# --------------------------------------------------------------------
# nwp.22 - DEBUG. Same, key read as a LOCATION.
# Console: event nwp.22
# --------------------------------------------------------------------
nwp.22 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		location:sevilla ?= {
			owner ?= { save_scope_as = cortes_sponsor }
		}
		location:cuetlaxtlan_province ?= {
			set_owner = scope:cortes_sponsor
			error_log = "::NWPCORTES::[GetCurrentYear]:DEBUG location-scope grant attempted"
		}
	}
}


# --------------------------------------------------------------------
# nwp.20 - the Cortes expedition. VISIBLE, with a choice.
#
# NOT HOOKED UP YET - the probes above must confirm the grant first. It is
# orphan = yes so it can be fired from the console to check the popup renders
# and the localization resolves, without waiting on the trigger.
#
# Two options, affirmative listed first so it is the player's default. The AI
# takes it at 100:0, so an observer run or an AI Castile always follows
# history while a human player keeps the veto.
#
# Declining sets the same done flag, so it asks once and then leaves you
# alone. Be aware that declining means nothing gives Mesoamerica a European
# foothold - which is precisely the untouched-Mexico outcome the 12-hour
# observer run produced, so the submod will have little to do there.
#
# If you would rather it keep asking every generation instead of once, drop
# the set_global_variable from the decline option and add a date check to the
# hook's limit.
#
# ADDITIONAL UNVERIFIED TOKENS: ai_chance, and the two scope branches below.
# Both grants are wrapped in ?= so whichever key form is wrong no-ops rather
# than firing. That handles a bad KEY. If `province:` or `location:` is not a
# valid scope TYPE at all it will be a load-time parse error instead, which
# shows up in error.log the moment the mod loads - check there first.
# --------------------------------------------------------------------
nwp.20 = {
	type = country_event
	title = nwp.20.t
	desc = nwp.20.d
	orphan = yes

	option = {
		name = nwp.20.a
		# EU5 has a game rule that makes the AI always take the option tagged
		# historical. The token for that tag could not be verified from this
		# repo, so it is commented rather than guessed - an unknown token
		# inside an option is a load-time parse error, and this file already
		# carries enough unverified surface.
		#
		# To find the real name, grep the base game's events for it:
		#   Select-String -Path "<EU5 game folder>\game\events\*.txt" -Pattern 'historical'
		# Then uncomment and correct the line below.
		#
		# historical = yes
		#
		# ai_chance stays regardless. With no option tagged historical the
		# engine falls back to it, so 100:0 produces the same behaviour the
		# historical rule would - the AI always sails. The only way the
		# expedition does not happen is a human player declining it, which is
		# the "history, with player intervention" model rather than random
		# alternate history every run.
		ai_chance = { factor = 100 }

		province:cuetlaxtlan_province ?= {
			every_location_in_province = {
				set_owner = root
			}
		}
		location:cuetlaxtlan_province ?= {
			set_owner = root
		}
		error_log = "::NWPCORTES::[GetCurrentYear]:expedition accepted, Veracruz granted"
	}

	option = {
		name = nwp.20.b
		ai_chance = { factor = 0 }

		error_log = "::NWPCORTES::[GetCurrentYear]:expedition declined"
	}
}

# a conquest of Andalusia in 1515 redirects the expedition; one in 1530 does
# not, which is the right way round.
