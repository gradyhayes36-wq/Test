namespace = nwp

# The Cortes expedition.
#
# WHY THIS EXISTS: in a 12-hour unsteered observer run the AI never took a
# single Mesoamerican location. Nothing in the submod is eligible to fire in a
# region no Old World power has reached, so the whole collapse model sat idle
# over the one region it was built for. EU5's Conquistador mechanic would
# normally cover this; M&T has it disabled.
#
# This forces the historical foothold - Veracruz, 1519, launched from the
# Caribbean - and lets the existing machinery take it from there. It is
# deliberately the SMALLEST possible intervention: one location, once. Nothing
# here conquers Mexico. It gives the beachhead and leaves the rest to the AI,
# the diseases and the region-contact sweep.
#
# THE SPONSOR IS WHOEVER OWNS SEVILLE, not a hardcoded Spain. Cortes sailed
# under Castile because Castile held Andalusia and the Casa de Contratacion;
# if some other power holds Seville in 1519, the expedition is theirs. This
# also removes the tag guess entirely - no c:SPA, no question of whether the
# Iberian union has formed yet - and it gives you the alternate history free:
# a Morocco that reconquered Andalusia inherits the conquest of Mexico.
#
# ============================ UNVERIFIED ============================
# These could not be checked against the repo's vanilla reference files,
# which contain only modifier definitions and situations:
#
#   1. set_owner              - the ownership-transfer effect name
#   2. seville                - Seville's key, and province: vs location:
#   3. cuetlaxtlan_province   - same question for Veracruz
#
# nwp.21 and nwp.22 settle all of it from the console. Do not enable nwp.20
# until one of them is confirmed working.
# ====================================================================


# --------------------------------------------------------------------
# nwp.21 - DEBUG. Both keys treated as PROVINCES.
# Console: event nwp.21
# --------------------------------------------------------------------
nwp.21 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		province:seville ?= {
			owner ?= {
				save_scope_as = cortes_sponsor
				error_log = "::NWPCORTES::[GetCurrentYear]:province-scope:sponsor resolved from Seville"
			}
		}
		province:cuetlaxtlan_province ?= {
			every_location_in_province = {
				set_owner = scope:cortes_sponsor
			}
			error_log = "::NWPCORTES::[GetCurrentYear]:province-scope:Veracruz grant attempted"
		}
	}
}


# --------------------------------------------------------------------
# nwp.22 - DEBUG. Both keys treated as LOCATIONS.
# Console: event nwp.22
# --------------------------------------------------------------------
nwp.22 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		location:seville ?= {
			owner ?= {
				save_scope_as = cortes_sponsor
				error_log = "::NWPCORTES::[GetCurrentYear]:location-scope:sponsor resolved from Seville"
			}
		}
		location:cuetlaxtlan_province ?= {
			set_owner = scope:cortes_sponsor
			error_log = "::NWPCORTES::[GetCurrentYear]:location-scope:Veracruz grant attempted"
		}
	}
}
