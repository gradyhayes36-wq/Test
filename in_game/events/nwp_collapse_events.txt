namespace = nwp

# Stamped when cocoliztli reaches a country, from the disease's own
# on_spread_to_country hook - see in_game/common/diseases/nwp_cocoliztli.txt.
#
# This hook lives INSIDE our own disease definition, so it needs no on_action
# entry and cannot collide with M&T's. Root scope is the country.
# Location sweep idiom copied from M&T's mnt_centers.1.
nwp.1 = {
	type = country_event
	title = hidden_event.t
	hidden = yes

	immediate = {
		set_local_variable = {
			name = nwp_stamped
			value = {
				every_owned_location = {
					limit = {
						continent = continent:america
						population > 1
						dominant_culture = { is_culture_native_american = yes }
						disease_presence = {
							disease = disease:cocoliztli
							value > 0.1
						}
					}
					add = 1
				}
			}
		}
		error_log = "::NWPLOG::[GetCurrentYear]:nwp.1 fired, stamping [SCOPE.GetLocalVariable('nwp_stamped').GetValue] locations"
		remove_local_variable = nwp_stamped

		every_owned_location = {
			limit = {
				current_date < 1600.1.1	# see the situation file
				continent = continent:america
				population > 1
				has_variable = nwp_contacted
				# Settler colonies are not what collapsed. The agricultural
				# landscape that stopped being maintained was the one the
				# natives built and worked, so this follows the population
				# rather than the map.
				dominant_culture = { is_culture_native_american = yes }
				disease_presence = {
					disease = disease:cocoliztli
					value > 0.1
				}
				# NO has_location_modifier filter here. It used to skip any
				# location already carrying the modifier, which made
				# `mode = replace` unreachable and meant a second cocoliztli
				# wave could never restart the 100-year clock - the exact
				# thing a recurring disease exists to do. Observed 1495:
				# "nwp.1 fired, stamping 0 locations", because the debug
				# stamp had already covered all 325.
			}

			# The maintainers are dead and the fields go with them. Long
			# duration: Mexico sat at the floor for roughly 150 years, and a
			# modifier measured in decades would expire well before 1700.
			add_location_modifier = {
				modifier = nwp_agricultural_collapse
				years = 60	# was 100
				mode = replace
			}

			# Under an Old World overlord, congregacion and forced labour
			# stack on top of the epidemic itself.
			if = {
				limit = { owner_from_old_world = yes }
				add_location_modifier = {
					modifier = nwp_colonial_labour_regime
					years = 60
					mode = replace
				}
			}

			# Chinampas, terraces and irrigation stop being maintained.
			# This is the mechanism, not a proxy for it: farming villages are
			# built one per 2,500 peasants at startup only (M&T mnt_food.1),
			# and the regeneration path (mnt_food.2) is gated on AI ownership,
			# monthly_income_total > 15 and >=800 unemployed pops - gates a
			# collapsed location plausibly fails.
			if = {
				limit = {
					has_building_with_at_least_one_level = farming_village
					disease_presence = {
						disease = disease:cocoliztli
						value > 0.4
					}
				}
				destroy_building = "building(building_type:farming_village|owner)"
			}
		}
	}
}

# Console-fired population census.
#
# M&T's own print_regional_population appears to emit nothing on 1.3 - a log
# with working ::GP:: and ::RT:: records had no ::POP:: at all, and no ::BT:
# either. Their print_country_information has its main ::TG:: line commented
# out, so these loggers do rot against game updates.
#
# This one is built only from patterns observed WORKING in that same log:
#   [GetCurrentYear]                             -> printed "1677"
#   [THIS.GetRegion.GetNameWithNoTooltip]        -> printed "Mesoamerica"
#   [SCOPE.GetLocalVariable('x').GetValue]       -> printed price values
# It deliberately avoids GetTotalPopulation, which is the one promote
# print_regional_population uses that this does not.
#
# orphan = yes so it can be fired from the console at will:
#     event nwp.9
# Then: Select-String "::NWPPOP::" error.log
nwp.9 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		every_region = {
			set_local_variable = {
				name = nwp_pop_total
				value = {
					every_ownable_location_in_region = {
						add = population
					}
				}
			}
			# Population as a fraction of capacity, averaged over inhabited
			# locations. This is the number that decides whether mortality can
			# ever work: at a fill ratio near 1.0 the region is pinned at its
			# soft cap, every death is refilled by growth, and only lowering
			# the cap changes anything.
			#
			# It is also the free-land bracket test. Vanilla's comment on
			# abundant_free_land reads "when its <10% capacity & <10k", and
			# M&T's bubonic plague uses location_population_percentage < 0.1
			# for the same low-density check.
			set_local_variable = {
				name = nwp_fill_sum
				value = {
					every_ownable_location_in_region = {
						limit = { population > 0 }
						add = location_population_percentage
					}
				}
			}
			set_local_variable = {
				name = nwp_loc_count
				value = {
					every_ownable_location_in_region = {
						limit = { population > 0 }
						add = 1
					}
				}
			}
			if = {
				limit = { local_var:nwp_loc_count > 0 }
				change_local_variable = {
					name = nwp_fill_sum
					divide = local_var:nwp_loc_count
				}
			}
			error_log = "::NWPPOP::[GetCurrentYear]:[THIS.GetRegion.GetNameWithNoTooltip]:[SCOPE.GetLocalVariable('nwp_pop_total').GetValue]:[SCOPE.GetLocalVariable('nwp_fill_sum').GetValue]:[SCOPE.GetLocalVariable('nwp_loc_count').GetValue]"
		}
		remove_local_variable = nwp_pop_total
		remove_local_variable = nwp_fill_sum
		remove_local_variable = nwp_loc_count

		# --- Status readout -------------------------------------------------
		# Turns a disappointing run into a diagnosis. Each counter isolates one
		# link in the chain, so a zero says exactly which part is not working:
		#   cocoliztli_locs = 0  -> the disease never spawned
		#   virgin_soil = 0      -> the Situation is not sweeping
		#   agri_collapse = 0    -> nwp.1 never fired, or its filter is too tight
		set_local_variable = {
			name = nwp_n_vs
			value = { every_ownable_location = { limit = { has_location_modifier = nwp_virgin_soil } add = 1 } }
		}
		set_local_variable = {
			name = nwp_n_ac
			value = { every_ownable_location = { limit = { has_location_modifier = nwp_agricultural_collapse } add = 1 } }
		}
		set_local_variable = {
			name = nwp_n_cl
			value = { every_ownable_location = { limit = { has_location_modifier = nwp_colonial_labour_regime } add = 1 } }
		}
		set_local_variable = {
			name = nwp_n_coc
			value = {
				every_ownable_location = {
					limit = { disease_presence = { disease = disease:cocoliztli value > 0 } }
					add = 1
				}
			}
		}
		# Every disease, not just ours. nwp_virgin_soil drops
		# local_disease_resistance by 0.75 across the Americas, which makes the
		# EXISTING diseases deadlier too - and malaria is endemic, environmental
		# and carries mortality { 0.85 0.95 }. If population falls before
		# cocoliztli has spawned, this line says which disease did it.
		set_local_variable = {
			name = nwp_n_mal
			value = { every_ownable_location = { limit = { disease_presence = { disease = disease:malaria value > 0 } } add = 1 } }
		}
		set_local_variable = {
			name = nwp_n_sp
			value = { every_ownable_location = { limit = { disease_presence = { disease = disease:smallpox value > 0 } } add = 1 } }
		}
		set_local_variable = {
			name = nwp_n_me
			value = { every_ownable_location = { limit = { disease_presence = { disease = disease:measles value > 0 } } add = 1 } }
		}
		set_local_variable = {
			name = nwp_n_fl
			value = { every_ownable_location = { limit = { disease_presence = { disease = disease:influenza value > 0 } } add = 1 } }
		}
		error_log = "::NWPSTAT::[GetCurrentYear]:virgin_soil=[SCOPE.GetLocalVariable('nwp_n_vs').GetValue]:agri_collapse=[SCOPE.GetLocalVariable('nwp_n_ac').GetValue]:labour_regime=[SCOPE.GetLocalVariable('nwp_n_cl').GetValue]:cocoliztli_locs=[SCOPE.GetLocalVariable('nwp_n_coc').GetValue]"
		error_log = "::NWPDIS::[GetCurrentYear]:malaria=[SCOPE.GetLocalVariable('nwp_n_mal').GetValue]:smallpox=[SCOPE.GetLocalVariable('nwp_n_sp').GetValue]:measles=[SCOPE.GetLocalVariable('nwp_n_me').GetValue]:influenza=[SCOPE.GetLocalVariable('nwp_n_fl').GetValue]"
		remove_local_variable = nwp_n_mal
		remove_local_variable = nwp_n_sp
		remove_local_variable = nwp_n_me
		remove_local_variable = nwp_n_fl
		if = {
			limit = { is_situation_active = situation:nwp_demographic_collapse }
			error_log = "::NWPSTAT::situation nwp_demographic_collapse ACTIVE"
		}
		else = {
			error_log = "::NWPSTAT::situation nwp_demographic_collapse NOT active"
		}
		remove_local_variable = nwp_n_vs
		remove_local_variable = nwp_n_ac
		remove_local_variable = nwp_n_cl
		remove_local_variable = nwp_n_coc

		error_log = "::NWPPOP::done"
	}
}

# =====================================================================
# Debug harness. Both are orphan = yes, so they fire from the console at
# any date against any save, and both bypass the contact gating entirely.
#
# The point is to test the two halves of the design SEPARATELY, in minutes,
# without colonising anything or running a campaign. If half B works and
# half A does not, that is a different fix from the reverse, and a long run
# with both active cannot tell them apart.
# =====================================================================

# HALF B: is cocoliztli lethal enough?
#   event nwp.7   then run ~10 years and compare nwp.9 census against baseline.
# Force-spawns at full presence in Mesoamerica, skipping monthly_spawn_chance
# and every spawn-block condition. spawn_disease in an effect context is the
# same idiom M&T's great_pestilence situation uses in its on_start.
nwp.7 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		# Matched control for nwp.5. Same spawn strength, same event shape,
		# same logging - the ONLY difference is the density of the seed
		# location. nwp.5 landed in a 2392-pop Caribbean location; this one
		# is held above 10 (ten thousand pops) so it lands in the agrarian core.
		region:mesoamerica_region = {
			random_ownable_location_in_region = {
				limit = { population > 10 }
				spawn_disease = {
					disease = disease:cocoliztli
					value = 1
				}
				set_local_variable = { name = nwp_dbg_pop value = population }
				error_log = "::NWPSPAWN::DEBUG seeded Mesoamerica:pop=[SCOPE.GetLocalVariable('nwp_dbg_pop').GetValue]"
				remove_local_variable = nwp_dbg_pop
			}
		}
	}
}

# HALF A: do the collapse modifiers actually suppress?
#   event nwp.6   then run ~20 years and compare nwp.9 census against baseline.
# Stamps every Mesoamerican location directly, skipping the Situation, the
# disease and the dominant_culture test. Isolates the modifiers' arithmetic
# from everything that has to work for them to be applied normally.
nwp.6 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		set_local_variable = { name = nwp_dbg value = 0 }
		region:mesoamerica_region = {
			every_ownable_location_in_region = {
				limit = { population > 0 }
				add_location_modifier = { modifier = nwp_virgin_soil days = -1 }
				add_location_modifier = {
					modifier = nwp_agricultural_collapse
					years = 60
					mode = replace
				}
				add_location_modifier = {
					modifier = nwp_colonial_labour_regime
					years = 60
					mode = replace
				}
				change_local_variable = { name = nwp_dbg add = 1 }
			}
		}
		error_log = "::NWPLOG::[GetCurrentYear]:DEBUG stamped collapse modifiers on [SCOPE.GetLocalVariable('nwp_dbg').GetValue] Mesoamerican locations"
		remove_local_variable = nwp_dbg
	}
}

# CONTROLLED TEST: does cocoliztli fail in thin populations?
#
# nwp.7 seeds Mesoamerica (~36k pops per location) and produced a wave that
# reached 243 of 325 locations. Every natural spawn has fizzled, and every
# natural spawn landed in a European-contacted location - which early on means
# the Caribbean at ~1.3k per location, or the East Coast at ~2.8k.
#
# Same disease, same value = 1, different density. Run both and compare
# cocoliztli_locs over the following decade. If the Caribbean seed dies while
# Mesoamerica explodes, density is the variable and the fix is to bias the
# spawn toward dense populations, not to make the disease stronger.
nwp.5 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	orphan = yes

	immediate = {
		region:caribbean_region = {
			random_ownable_location_in_region = {
				limit = { population > 1 }
				spawn_disease = {
					disease = disease:cocoliztli
					value = 1
				}
				set_local_variable = { name = nwp_dbg_pop value = population }
				error_log = "::NWPSPAWN::DEBUG seeded Caribbean:pop=[SCOPE.GetLocalVariable('nwp_dbg_pop').GetValue]"
				remove_local_variable = nwp_dbg_pop
			}
		}
	}
}
