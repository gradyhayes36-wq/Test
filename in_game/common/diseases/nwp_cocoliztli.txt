# Cocoliztli - https://en.wikipedia.org/wiki/Cocoliztli_epidemics
# New disease, additive. No REPLACE:/INJECT: prefix because nothing in vanilla
# or M&T defines this key.
#
# Field set and idiom copied from M&T's in_game/common/diseases/MnT_smallpox.txt.
# The non-immunising, high-mortality shape is copied from MnT_malaria.txt.
#
# Design note: this deliberately does NOT hook the great_pestilence Situation.
# M&T's Situation has no disease registry - the three diseases gate their own
# spawn blocks on the Situation's state, and the Situation only force-spawns
# smallpox in on_start. So a fourth disease needs no registration at all, and
# by gating on its own date window it outlives the Situation for free.

cocoliztli = {

	# ~1 wave per 20 years while the window is open. See nwp_tuning.txt.
	monthly_spawn_chance = {
		value = 0
		if = {
			limit = {
				# Window opens once Old World contact exists in the Americas,
				# and stays open long after the Great Pestilence Situation ends.
				continent:america = {
					any_location_in_continent = {
						population > 0
						owner_from_old_world = yes
					}
				}
			}
			value = nwp_cocoliztli_monthly_spawn_chance
		}
	}

	spawn = {
		random_ownable_location = {
			limit = {
				population > 1
				continent = continent:america

				# Densely settled, agrarian Mesoamerica and the Andes took the
				# worst of it. Cocoliztli is not a frontier disease.
				OR = {
					tropical_climate_trigger = yes
					subtropic_climate_trigger = yes
					mediterranean_climate_trigger = yes
					continental_climate_trigger = yes
				}

				# Only where the colonial regime has actually arrived: the
				# 1545 catastrophe followed congregacion, forced labour and
				# the collapse of the maintained agricultural landscape.
				OR = {
					owner_from_old_world = yes
					has_variable = had_great_pestilence_in_location
					location_has_pestilence_trigger = yes
				}

				# And only where there are natives to kill. Vanilla's
				# great_pestilence gates its spawn the same way, with
				# any_pop = { culture = { is_culture_native_american = yes } }.
				# A settler colony is not a cocoliztli target.
				any_pop = {
					culture = { is_culture_native_american = yes }
				}
			}
			spawn_disease = {
				disease = scope:disease
				value = nwp_cocoliztli_spawn_strength
				save_scope_as = outbreak
			}
		}
	}

	# Enteric fever: spread by contaminated water and crowding rather than
	# breath, so denser settlement matters more than contact frequency.
	r0 = {
		if = {
			limit = {
				location_rank ?= location_rank:rural_settlement
			}
			value = { 2 4 }
		}
		else_if = {
			limit = {
				location_rank ?= location_rank:town
			}
			value = { 4 8 }
		}
		else = {
			value = { 8 12 }
		}

		# Congregacion concentrated survivors into new settlements with
		# water supplies nobody had built for that many people.
		multiply = {
			value = "disease_presence(scope:disease)"
			multiply = 2
			add = 1
		}
	}

	calc_interval_days = { 6 14 }

	location_stagnation_chance = {
		if = {
			limit = {
				"disease_presence(scope:disease)" > 0.05
			}
			value = "distance_to_squared(scope:disease.origin)"
			multiply = 0.0006
			multiply = 0.0035
		}
		else = {
			value = "distance_to_squared(scope:disease.origin)"
			multiply = 0.00004
		}
	}

	# Calibration note. Lethality is not the mortality_rate alone - it is
	# mortality x how long the disease sits at high presence. Vanilla's
	# great_pestilence pairs mortality { 0.75 0.9 } with a fate rate of only
	# 0.10, so presence saturates and stays saturated. M&T's smallpox burns
	# through at 0.25 and collapses. We want great_pestilence's shape at
	# roughly half its severity, because ours recurs and its does not.
	percentage_to_meet_their_fate_on_calc = {
		value = nwp_cocoliztli_fate_per_calc
	}

	# Vanilla great_pestilence 0.60 (0.85 in hard terrain), smallpox 0.15,
	# bubonic 0.30. High thresholds keep the disease concentrated and lethal
	# rather than smeared thinly across the map.
	location_spread_threshold = {
		if = {
			limit = {
				num_roads = 0
				OR = {
					topography = mountains
					vegetation = jungle
				}
			}
			value = 0.55
		}
		else = {
			value = 0.40
		}
	}

	# Half A, the suppression half, in its cheapest form: while a wave is
	# present the location cannot grow back. Scaled by presence by the engine.
	# The PERSISTENT century-long collapse is a separate modifier - see
	# main_menu/common/static_modifiers/nwp_collapse.txt. It is not wired up
	# yet; that needs an on_action or event hook. See README.
	location_modifier = {
		# Vanilla great_pestilence uses -0.1 here; M&T's bubonic plague omits
		# growth entirely and leans on attrition. -0.05 is half the vanilla
		# apocalypse, which is the right place for a disease that returns.
		# Scaled by presence by the engine, so this is the value at full
		# saturation, not a flat penalty.
		local_population_growth = nwp_collapse_growth_penalty
		local_prosperity_decay = 0.3
		local_monthly_development = -0.02
		local_migration_attraction = -20.0
		local_forced_attrition = 5
		max_attrition = 10
		local_pop_promotion_speed_modifier = -2.0
		local_pop_demotion_speed_modifier = 2.0

		# Fields go unworked during a wave. This is also what pushes a
		# location toward vanilla's province_starving, which carries
		# local_population_growth = -0.025 - far heavier than anything
		# worth writing by hand here.
		local_monthly_food_modifier = -0.3
	}

	# The 1576 wave killed survivors of 1545. Resistance must not carry over
	# between waves the way smallpox resistance does (0.0007/month there).
	monthly_resistance_reduction = nwp_cocoliztli_resistance_decay

	# Vanilla great_pestilence { 0.75 0.9 }, malaria { 0.85 0.95 },
	# bubonic { 0.3 0.6 }, smallpox { 0.05 0.3 }.
	# Cocoliztli killed an estimated 5-15M in 1545-48 alone, but it recurs
	# here, so per-wave severity sits below the one-shot apocalypse.
	mortality_rate = {
		value = { 0.4 0.6 }
	}

	character_mortality_chance = {
		value = 0.06
		multiply = "disease_presence(scope:disease)"
	}

	# The hook. This lives inside our own disease definition, so it needs no
	# on_action entry and cannot collide with M&T's lists. Root is the
	# country; nwp.1 sweeps its American locations and stamps the persistent
	# collapse modifiers. See in_game/events/nwp_collapse_events.txt.
	on_spread_to_country = {
		trigger_event_silently = nwp.1
	}

	map_color = {
		lerp = { # Ochre
			min_color = rgb { 214  178  110 }
			max_color = rgb { 110 62 6 }
			factor = "disease_presence(scope:disease)"
		}
	}

	secondary_map_color = {
		if = {
			limit = {
				disease_affects_pops_here = scope:disease
			}
			value = rgb { 110 62 6 }
		}
	}
}
